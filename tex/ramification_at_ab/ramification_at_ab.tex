  \section{Local Analysis}
  Let $\F$ be a non-Archimedean local field with local ring $R$, and residue field $\kappa$. 
  Let $\overline{\F}$ be an algebraic closure of $\F$ and let $\overline{\kappa}$ be the residue field of $\overline{\F}$.
  Let $v$ be an extension to $\overline{\F}$ of the unique valuation of $\F$ for which $v\left( K \right) = \Z$.
  % Fix $n,m,k$ pairwise distinct non-negative integers with $\gcd\left( n,m,k \right) = 1$.
  Let 
  \[
    \h{} = a_n x^n + a_m x^m + a_k x^k \in \F\left[ x, x^{-1} \right]
  \]
  be a trinomial with $a_n, a_m, a_k \neq 0$ and 
  \[
    1 = \gcd\left( n - m, m - k, k - n \right) = \gcd\left( n - m, m - k \right).
  \]
  When use $\left( \h{} \right)$ to denote the principal ideal generated by $\h{}$ in the ring $\kappa\left[ x, x^{-1} \right]$.
  Note that no assumption is made about the order of $n,m,k$ and thus all statements about them are true for all of the permutations of $n,m,k$.
  In light of this, in the sequel we often state or define something for some permutaion of $n,m,k$ and then use it for other permutations as well. 
  
  Let $\alpha \in \overline{\F}^\times$.
  Note that if $\alpha$ is a root of $\h{}$, then
  \[
    \alpha \h{}'\left( \alpha \right) 
    = 
    \alpha \h{}'\left( \alpha \right) - n\h{}\left( \alpha \right)
    % n a_n \alpha^n + m a_m \alpha^m + k a_k \alpha^k 
    =
    (m - n) a_m \alpha^m + (k - n) a_k \alpha^k.
  \] 
  Thus, let us denote
  \[
    \h{n} 
    =
    x\h{}' - n\h{} 
    = 
    (m - n) a_m x^m + (k - n) a_k x^k
  \]
  For most cases, we can bound $v\left( \h{}'\left( \alpha \right) \right)$ using the non-Archimedean triangle inequality and the summation term of least valuation of either $\h{n}$, $\h{m}$ or $\h{k}$.
  We will try to first characterize the cases where this strategy fails.
  
  \subsection{Magnetic roots}
  So we can speak of each of the summation terms of $\h{n}$ separately, let us denote
  \[
    \h{n,m} = (m - n) a_m x^m.
  \]
  Note that 
  \begin{align}
    \label{eq:valuation_inequality_condition}
    & v\left( \alpha\h{n}\left( \alpha \right) \right) 
    > 
    \min\left\{  
      v\left( \h{n,m}\left( \alpha \right) \right), 
      v\left( \h{n,k}\left( \alpha \right)\right)
    \right\}
  \end{align}
  if and only if
  \begin{equation}
    \label{eq:valuation_positivity_condition}
    v\left( 1 + \frac{\h{n,m}\left( \alpha \right)}{\h{n,k}\left( \alpha \right)} \right) > 0,
  \end{equation}
  \directorscut{(indeed, in both cases $v\left( \h{n,k}\left( \alpha \right) \right) = v\left( \h{n,m}\left( \alpha \right) \right)$, and hence the equivalence follows by reducing $v\left( \h{n,k}\left( \alpha \right) \right)$ from both sides of the first inequality),}
  i.e. if and only if 
  \begin{equation}
    \label{eq:principal_unit_condition}
    \q{n}{m} :=
    -\frac{\h{n,m}\left( \alpha \right)}{\h{n,k}\left( \alpha \right)}
    =
    -\frac{
      \left( m - n \right)a_m \alpha^m
    }{
      \left( k - n \right)a_k \alpha^k
    }
    \text{ is a principal unit.}
  \end{equation}
  To state the last condition in a more useful language, we make the following
  \begin{definition}
    \label{def:pi}
    Let $\pu$ be the groups of principal units in $\overline{\F}$, i.e the groups of elements in the respective fields whose difference from $1$ has positive valuation. 
    Let 
    \[
      \pi: \overline{\F}^\times \to \frac{\overline{\F}^\times}{\pu}
    \]
    be the canonical quotient map. 
  \end{definition} 
  In this language condtion \ref{eq:principal_unit_condition} can be stated as
  \begin{equation}
    \label{eq:mod_principal_units_condition}
    \pi\left(  
      \q{n}{m}
    \right) = 
    \pi\left( \alpha^{k - m} \right)
  \end{equation}
  Since $\F\left[ x, x^{-1} \right]^\times = \F^\times x^{\Z}$, if condition \ref{eq:mod_principal_units_condition} holds for $h{}$ then it also holds for any polynomial in the ideal $\left( \h{} \right)$.
  Thus, we can make the following
  \begin{definition}
    If the equivalent conditions \ref{eq:valuation_inequality_condition}, \ref{eq:valuation_positivity_condition}, \ref{eq:principal_unit_condition} and \ref{eq:mod_principal_units_condition} hold, we say that $\pi\left( \alpha \right)$ is an \textbf{\es{n}} disk of $\left( \h{} \right)$.
    % , or just \es{n} when $\left( \h{} \right)$ is clear from context
    We say that $\pi\left( \alpha \right)$ is \textbf{\anex} disk of $\left( \h{} \right)$
    % or just \anex root when $\h{}$ is clear from context
    if it is \es{n}, \es{m} and \es{k}.
    In many cases we write $\h{}$ instead of $\left( \h{} \right)$, and if $\alpha$ is also a root of $\h{}$ we say that it is \textbf{\anex (respectively an \es{n}) root of $\h{}$}.
  \end{definition}
  We use the analogy of magnetism because, as explained above, for a magnetic root $\alpha$ $v\left( \h{}'\left( \alpha \right) \right)$ is bigger than, i.e., in the metric induced by $v$, $\alpha$ is on avarage closer to its conjugates than one might expect by bounding the valuations of $\h{i,j}$ for each pair of distinct $i,j \in \left\{ m,n,k \right\}$.
  We use the term disk because each fiber of $\pi$ is a disk in the metric induced by $v$. 

  \begin{lemma}
    \label{lem:scalability}
    Assume that $\pi\left( \alpha \right)$ is \anex disk of $\h{}$.
    Fix $a \in \F^\times$ and $\delta =\pm 1$.
    Then $\pi\left( a\alpha^\delta \right)$ is \anex disk of $\h{}\left( x^\delta / \alpha \right)$.
  \end{lemma}
  \begin{proof}
    Let $u\left( x \right)$ be $a x^\delta $.
    Then $\h{} \circ u$ is a trinomial with coefficients $\delta m, \delta n, \delta k$ and.
    One can check that 
    \[
      \left( \h{} \circ u^{-1} \right)_{\delta m,\delta n}\left( u\left( \alpha \right) \right) = \left( \h{} \right)_{m,n}\left( \alpha \right)
    \]
    and the rest follows from opening the definition of \ex disks.
  \end{proof}

  \begin{corollary}
    \label{cor:ex_disk}
    There are $r,s \in \Z$ depenfing only on $n-m$ and $m-k$, such that $\pi\left( \alpha \right)$ is \anex disk of $\h{}$ if and only if it solves the $\Z$-linear equation
    \begin{equation}
      \label{eq:ex}
      \begin{bmatrix}
        1 \\
        0
      \end{bmatrix} x 
      = 
      \begin{bmatrix}
        \pi\left( \q{n}{m}^r \q{m}{k}^s \right) \\
        \pi\left( \q{n}{m}^{-\left( m - k \right)} \q{m}{k}^{n - m} \right)
      \end{bmatrix} .
    \end{equation}
  \end{corollary}
  \begin{proof}
    Another way to state Lemma \ref{lem:two_escaping_implies_ex} is that $\pi\left( \alpha \right)$ is \anex disk of $\h{}$ if and only if it solves the $\Z$-linear equation
    \begin{equation}
      \label{eq:raw_ex_equation}
      \begin{bmatrix}
        n - m \\
        m - k
      \end{bmatrix} x 
      = 
      \begin{bmatrix}
        \pi\left( \q{n}{m} \right) \\
        \pi\left( \q{m}{k} \right)
      \end{bmatrix}.
    \end{equation}
    Since by assumption $n - m$ and $m - k$ are coprime, by the Euclidean algorithm there are $r,s \in \Z$ such that 
    \[
      r(n - m) + s(m - k) = 1.
    \]
    Therefore, multiplying both sides of equation \ref{eq:raw_ex_equation} by 
    \[
      \begin{bmatrix}
        r & s \\
        -\left( m - k \right) & n - m
      \end{bmatrix}
    \]
    shows that it is equivalent to the required equation.
  \end{proof}
  
  \begin{lemma}
    \label{lem:reduction}
    If $\pi\left( \alpha \right)$ is \anex disk of $\h{}$, then there is a $u \in \F^\times x^{\pm 1}$ such that : 
    \begin{enumerate}
      \item 
      $\pi\left( 1 \right)$ is \anex disk of $\h{} \circ u^{-1}$.
      % \item 
      % $u\left( \alpha \right)$ is \anex root of $u\left( \h{} \right)$.
      \item 
      $\h{} \circ u^{-1}$ is integral over $R$.
    \end{enumerate}
  \end{lemma}
  \begin{proof}
    According to Corollary \ref{cor:ex_disk}, $\pi\left( \alpha \right) = \q{n}{m}^r \q{m}{k}^s$ when $r$ and $s$ are as defined in Corollary \ref{cor:ex_disk}.
    Set $q = \q{n}{m}^r \q{m}{k}^s$.
    Therefore, $\pi\left( q^{-1}\alpha \right) = 1$ and in particular $v\left( q^{-1}\alpha \right) = 0$.
    Since $\h{}\left( qx \right)$ is a trinomial, its Newton polygon has at most two segments, and hence the set of valuations of its roots is of cardinality at most $2$.
    Hence, as we have already shown that $0$ is in the set of valuations of roots of $\h{}$, it follows that this set is either non negative or non positive.
    We can set $\delta = 1$ in the first plave and $\delta = -1$ in the second place (and either of them if both cases hold) and set $u = q x^\delta$.
    Then $\h{} \circ u^{-1}$ is integral over $R$ and $\pi\left( 1 \right)$ is \anex disk of $\h{} \circ u^{-1}$ according to Lemma \ref{lem:scalability}.
  \end{proof}
  
  \begin{lemma}
    \label{lem:ex_lift}
    If $\h{}$ has \anex disk then it has \anex root.  
  \end{lemma}
  \begin{proof}
    By Lemma \ref{lem:reduction} we can assume without loss of generality that $\h{}$ is integral over $R$ and that $\pi\left( 1 \right)$ is \anex disk of it.
    In this situation, it suffices to show that $1$ lifts to a root of $\h{}$, or equivalently, by the integrality of $\h{}$, that $\h{}\left( 1 \right) = 0 \in \kappa$.
    Indeed, by condition \ref{eq:mod_principal_units_condition}
    \begin{align*}
      \h{}\left( 1 \right) 
      = 
      & \
      a_n + a_m + a_k
      =
      a_n + a_m \q{k}{n} + a_k \q{m}{n}
      =
      % \\
      % &
      a_n - \frac{\left( n - k \right)a_n}{\left( m - k \right)} - \frac{\left( n - m \right)a_n}{\left( k - m \right)} 
      = 
      0
      .
    \end{align*}
  \end{proof}

  \subsection{Reduced magnetic roots}
  In light of Lemma \ref{lem:reduction} we make the following
  \begin{definition}
    \label{def:aredex}
    We say that $\alpha$ ia \textbf{\aredex root} of $\h{}$ and that $\h{}$ is \\ \textbf{\aredex polynomial} if:
    \begin{enumerate}
      \item $\alpha$ is \anex root of $h$.
      \item $\pi\left( \alpha \right) = 1$. 
      \item $\h{}$ is intagral and primitive over $R$.
    \end{enumerate}
  \end{definition} 
  
  Note that if $\alpha$ is \aredex root of $\h{}$, then the Newton polygon of $\h{}$ has a segment of slope $0$.
  Therefore, no loss of generality arrieses from replacing the condition that $\alpha$ is \anex root of $\h{}$ with the condition of the following
  \begin{lemma}
    \label{lem:slope_length_valuation}
    Assume that $\alpha$ is \aredex root of $\h{}$ and that the segment connecting the points $\left( k, v\left( a_k \right) \right)$ and $\left( m, v\left( a_m \right) \right)$ belong to the Newton polygon of $\h{}$ and has slope $0$.
    Then
    \begin{enumerate}
      \item \label{item:smaller_slope_length_valuation} 
      $v\left( n - k \right) = v\left( n - m \right) = 0$.
      \item \label{item:bigger_slope_length_valuation} 
      $v\left( m - k \right) 
        = 
        v\left( a_n \right)
        %  - v\left( a_m \right)
        % =
        % v\left( a_n \right) - v\left( a_k \right)
        .$
    \end{enumerate}
  \end{lemma}
  \begin{proof}
    Condition \ref{eq:valuation_inequality_condition} in the definition of $k$-magnetism and $ m $-magnetism implies the equalities 
    \begin{equation}
      \label{eq:raw_valuation_equality}
      v\left( \h{k,n}\left( \alpha \right) \right) = 
      v\left( \h{k,m}\left( \alpha \right) \right)
      \quad \text{and} \quad
      v\left( \h{m,n}\left( \alpha \right) \right) = 
      v\left( \h{m,k}\left( \alpha \right) \right)
      .
    \end{equation}
    To simplify it, note that by our assumptions regarding the Newton polygon of $\h{}$ it follows that $v\left( a_n \right) > v\left( a_m \right) = v\left( a_k \right)$, and that since we also assume that $\h{}$ is \aredex polynomial and thus intagral and primitive over $R$, it follows that $v\left( a_m \right) = v\left( a_k \right) = 0$. 
    Substituting these two facts into equation \ref{eq:raw_valuation_equality}, as well as the assumption that $v\left( \alpha \right) = 0$, gives the equalities
    \begin{gather}
      v\left( m - k \right) = v\left( n - k \right) + v\left( a_n \right) 
      \label{eq:first_relation_between_slope_length_valuations}
      \\
      \notag
      \text{ and } 
      \\
      v\left( m - k \right) = v\left( n - m \right) + v\left( a_n \right) 
      \label{eq:second_relation_between_slope_length_valuations}
    \end{gather}
    which together imply that $v\left( n - k \right) = v\left( n - m \right)$.
    Since $1 = \gcd\left( n - m, n - k \right)$, it hence follows that $v\left( n - k \right) = v\left( n - m \right) = 0$, proving item \ref{item:smaller_slope_length_valuation}.
    Substituting back in either equation \ref{eq:first_relation_between_slope_length_valuations} or equation \ref{eq:second_relation_between_slope_length_valuations} proves item \ref{item:bigger_slope_length_valuation}.
  \end{proof}

  \subsubsection{The equal valuation case}
  Assume now that $v\left( a_n \right) = v\left( a_m \right) = v\left( a_k \right) = 0$ and that $\h{}$ is \aredex polynomial.

  \begin{lemma}
    \label{lem:magnetism_away_from_ab}
    $\alpha$ is \anex root of $\h{}$ if and only if $v\left( \h{}'\left( \alpha \right) \right) > 0$.
  \end{lemma}
  \begin{proof}
    Since the Newton polygon of $\h{}$ has only segments and its slope is $0$, $v\left( \alpha \right) = 0$.
    Therefore, since $1 = \gcd\left( m - n, k - n \right)$, it follows that 
    \[
      \min\left\{  
        v\left( \h{n,m}\left( \alpha \right) \right),
        v\left( \h{n,k}\left( \alpha \right) \right)
      \right\} = 0
    \]
    and hence the result follows from condition \ref{eq:valuation_inequality_condition} defining magnetism.
  \end{proof}

  \begin{lemma}
    \label{lem:2_magnetic_roots}
    $\h{}$ has at most $2$ magnetic roots.
  \end{lemma}
  \begin{proof}
    % Assume that it has \anex root.
    % Then by Lemma \ref{lem:scalability} and Lemma \ref{lem:reduction} we can assume that it has \anex above $1$, and hence, 
    By Corollary \ref{cor:ex_disk} all of its \ex roots lie above $1$.
    Therefore, it suffices to show that $1 \in \kappa$ is not a triple root of $\h{}$.
    However if it was it would have been a double root of 
    \[
      \h{n} 
      = 
      x\h{}' - n\h{} 
      =
      x^k \left( a_m x^{m - k} + a_k \right) 
    \] 
    contradicting the reducedness of $\left( a_m x^{m - k} + a_k \right) \leq \kappa\left[ x, x^{-1} \right]$ which follows form the fact that $v\left( m - k \right) = 0$ which follows in turn from Lemma \ref{lem:slope_length_valuation} whose conditions we assume above.
  \end{proof}

  \begin{corollary}
    The inertia group acts on the roots of $\h{}$ either trivially or as a single transposition.
  \end{corollary}
  \begin{proof}
    It suffices to show that $\h{}$ has at most $2$ roots $\alpha$ at which $v\left( \h{}'\left( \alpha \right) \right) > 0$, or equivalently, according to Lemma \ref{lem:magnetism_away_from_ab}, to show that $\h{}$ has at most two \ex roots, which is true by Lemma \ref{lem:2_magnetic_roots}. 
  \end{proof}
  \begin{corollary}
    If $v_p\left( ab \right) = 0$ for any rational prime $p$, then the inertia group at any place above $p$ acts on the roots of $f$ either trivially or as a single transposition.
  \end{corollary}

    Therefore, since we assume here that $\h{}$ is \aredex polynomial and in particular integral over $R$, it follows that $\h{}$ has exactly $p^{v\left( m - k \right)}$ roots in $\overline{\F}$ lying over $1$, which means, according to Corollary \ref{cor:ex_disk}, that $\h{}$ has exactly $p^{v\left( m - k \right)}$ \ex roots.
  \end{proof}

  Let $p$ be a rational prime.
  We denote by $\D_\alpha$ the different ideal of $\Q_p\left( \alpha \right)$ for each $\alpha \in \overline{\Q_p}$ and deduce from Lemma \ref{lem:counting_ex_roots} the following

  \begin{corollary}\label{cor:ex_bound}
    Assume that $p| ab$.
    Then $f$ has at most $\ppart{ab}$ \ex roots above $\Q_p$ and for each such root $\alpha$, 
    \[
      v_p\left( \Deta \right) \leq \diffbnd.
    \]
  \end{corollary}
  \begin{proof}
    Let $\alpha$ be \anex root of $f$ (if there is no such root then the result is trivial).
    According to Corollary \ref{cor:ex_disk}, $\pi\left( \alpha \right)$ is the only \ex disk and it belongs to $\pi\left( \Q^\times \right)$ (recall Definition \ref{def:pi} for $\pi$).
    In particular, the valuation of the \ex roots is a (unique) integer.
    If that integer is at least $1$, then the number of \ex roots can not exceed the maximal height difference between two vertices of the Newton polygon of $f$, which is clearly less than $\exrootscountbound$.
    Assume then that the valuation of the \ex roots is $0$.
    Then, by Lemma \ref{lem:scalability} we can assume without loss of generality that $f$ is \aredex polynomial.
    In particular exactly one of the coefficients $a$ and $b$ has positive valuation, and according to Lemma \ref{lem:counting_ex_roots}, the number of \ex roots of $f$ is bounded by $p$ raised to the valuation of that coefficient, which is exactly $\exrootscountbound$.

    In particular it follows that $\alpha$ has at most $\exrootscountbound$ conjugates above $\Q_p$ and hence its ramification degree is at most $\exrootscountbound$.
    Denote this ramification degree by $e$.
    Then, $1-e^{-1} + v_p\left( e \right)$ is a well known upper bound on the valuation of $\Deta$, the different of $\Q_p\left( \alpha \right)$ over $\Q_p$ (see for example \cite[Remark 1 to Proposition 13 of Chapter. III \S 7]{SerreLocalFields}).
    However, $$1 - e^{-1} + v_p\left( e \right) \leq 1 + v_p\left( e \right) \leq 1 + v_p\left( p^{v_p\left( ab \right)} \right) = 1 + v_p\left( ab \right)$$ and hence the result follows.
  \end{proof}

  \subsection{The derivative at non \ex roots}
  \begin{lemma}
    \label{lem:non_es_bound}
    If $\alpha$ is a non \es{n} root of $\h{}$ then
    \[
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq
      \max\left\{ a_m \alpha^m, a_k \alpha^k \right\} 
    \]
  \end{lemma}
  \begin{proof}
    From the definition of \es{n} roots it follows that 
    \[
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq
      \min\left\{ \left( m - n \right)a_m \alpha^m, \left( k - n \right)a_k \alpha^k \right\}
    \]
    and hence the result follows from the assumption that 
    $
      1 = \gcd\left( m - n, k - n \right)
    $
    and hence either $v\left( m-n \right) = 0$ in which case $v\left( \alpha \h{}'\left( \alpha \right) \right) \leq v\left( a_m \alpha^m \right)$ or $v\left( k - n \right) = 0$ in which case $v\left( \alpha \h{}'\left( \alpha \right) \right) \leq v\left( a_k \alpha^k \right)$.
  \end{proof}

  \begin{corollary}
    \label{cor:non_ex_bound}
    If $\alpha$ is a non \ex root of $\h{}$ then
    \[
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq
      \max\left\{  
        v\left( a_n \alpha^n \right),
        v\left( a_m \alpha^m \right),
        v\left( a_k \alpha^k \right)
      \right\}.
    \]
  \end{corollary}
  \begin{proof}
    By Lemma \ref{lem:two_escaping_implies_ex} we can assume without loss of generality that $\alpha$ is neither an \es{n} root of $\h{}$ nor an \es{m} root of $\h{}$.
    Therefore, it follows from Lemma \ref{lem:non_es_bound} that 
    \begin{align*}
      v\left( \alpha \h{}'\left( \alpha \right) \right)
      \leq
      &
      \min\left\{  
        \max\left\{  
          a_m \alpha^m,
          a_k \alpha^k
        \right\},  
        \max\left\{  
          a_n \alpha^n,
          a_k \alpha^k
        \right\}
      \right\}
      \leq \\
      & \max\left\{  
        \max\left\{  
          a_m \alpha^m,
          a_k \alpha^k
        \right\},  
        \max\left\{  
          a_n \alpha^n,
          a_k \alpha^k
        \right\}
      \right\} \leq \\
      & \max\left\{  
        a_n \alpha^n,
        a_m \alpha^m,
        a_k \alpha^k
      \right\}
      .
    \end{align*} 
  \end{proof}

  Just like in the case of Lemma \ref{lem:slope_length_valuation}, the condition that $\alpha$ is a non \ex root of $\h{}$ is just as general as the condition of the following
  \begin{corollary}
    \label{cor:explicit_non_ex_bound}
    If $\alpha$ is a non \ex root of $\h{}$, and the segment connecting the points $\left( m, v\left( a_m \right) \right)$ and $\left( n, v\left( a_n \right) \right)$ belong to the Newton polygon of $\h{}$ and its slope is $-v\left( \alpha \right)$, then
    \begin{gather}
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq
      v\left( a_k \right) + \frac{k}{n - m}v\left( \frac{a_m}{a_n} \right)
      \label{item:k_es}
    \end{gather}
    and 
    \begin{gather}
      \text{either }
      \quad
      v\left( n - m \right) 
      >
      0
      \quad
      \text{ or }
      \quad
      v\left( \alpha \h{}'\left( \alpha \right) \right)
      \leq 
      \frac{
        n v\left( a_m \right) - m v\left( a_n \right)
      }{
        n - m
      }.
      \label{item:non_k_es}
    \end{gather}
  \end{corollary}
  \begin{proof}
    For each $j \in \left\{ k,n,m \right\}$, $v\left( a_j \alpha^j \right)$ is the $y$-coordinate of the intersection with the $y$-axis of the line of slope $-v\left( \alpha \right)$ passing through the point $\left( j, v\left( a_j \right) \right)$.
    Since the Newton polygon of $\h{}$ contains the segment connecting the points $\left( m, v\left( a_m \right) \right)$ and $\left( n, v\left( a_n \right) \right)$, the point $\left( k, v\left( a_k \right) \right)$ lies above the line containing this segment.
    However, the slope of this line is $-v\left( \alpha \right)$, and hence among the lines of slope $-v\left( \alpha \right)$ passing through the points $\left( j, v\left( a_j \right) \right)$ for $j \in \left\{ k,n,m \right\}$, the one passing through $\left( k, v\left( a_k \right) \right)$ has the largest intersection with the $y$-axis.
    Therefore, according to Corollary \ref{cor:non_ex_bound}, $v\left( \alpha \h{}'\left( \alpha \right) \right) \leq v\left( a_k \alpha^k \right)$, and it can be verified that $v\left( a_k \alpha^k \right)$ equals to the bound presented in condition \ref{item:k_es}.

    To prove condition \ref{item:non_k_es}, assume that $v\left( n - m \right) = 0$ and we will prove that the required bound on $v\left( \alpha \h{}'\left( \alpha \right) \right)$ holds.
    Since $\alpha$ is assumed to be non-magnetic, according to Lemma \ref{lem:two_escaping_implies_ex} it is either \es{m} or \es{n}.
    By the criterion \ref{eq:valuation_inequality_condition} for \es{n}, it follows that in the first case 
    $$
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq 
      v\left( \h{m,n}\left( \alpha \right) \right)
      =
      v\left( \left( n - m \right) a_n \alpha^n \right)
      =
      v\left( a_n \alpha^n \right)
    $$
    and in the second case
    $$
      v\left( \alpha \h{}'\left( \alpha \right) \right) 
      \leq 
      v\left( \h{n,m}\left( \alpha \right) \right)
      =
      v\left( \left( m - n \right) a_m \alpha^m \right)
      =
      v\left( a_n \alpha^n \right)
    $$
    and it can be verified that both $v\left( a_n \alpha^n \right)$ and $v\left( a_m \alpha^m \right)$ are equal to the bounf on $v\left( \alpha\h{}'\left( \alpha \right) \right)$ presented in condition \ref{item:non_k_es}.
  \end{proof}
  \subsection{The discriminant}
  Fix a rational prime $p$.
  Recall that we denote by $g$ an arbitrary factor of $f$, by $L$ the field $\Q\left[ x \right] / \left( g \right)$ and by $\Deta$ the different ideal of the image of $L$ in $\overline{\Q_p}$ under the embedding that maps $x$ to $\alpha$.
  \begin{proposition}
    Assume that $p | ab$.
    Then
    \[
      v_p\left( \disc\left( L \right) \right)
      \leq
      nv_p\left( ab \right) + \exrootscountbound.
    \]
    % and if $\deg\left( g \right) > \exrootscountbound$ then $g$ has at least one root $\alpha$ for which 
    % \[
    %   v_p\left( \Deta \right) < v_p\left( ab \right).
    % \]
  \end{proposition}
  \begin{proof}
    We divide now into three cases.
    Assume first that the Newton polygon of $f$ consists of a single segment.
    Then, by condition \ref{item:k_es} of Corollary \ref{cor:explicit_non_ex_bound}, for each non \ex root $\alpha$ of $f$,
    \[
      v_p\left( \Deta \right) 
      \leq
      v_p\left( \alpha f'\left( \alpha \right) \right)
      \leq
      v_p\left( a \right) + \frac{k}{n} v_p\left( b \right)
      \leq
      v_p\left( ab \right)
    \]
    and therefore by Corollary \ref{cor:ex_bound},    
    \begin{align*}
      v_p\left( \disc\left( L \right) \right)
      =
      &
      \sum_{\alpha} v_p\left( \Deta \right)
      \leq
      \left( n - \exrootscountbound \right) v_p\left( ab \right)
      +
      \exrootscountbound \left( \diffbnd \right)
      \leq
      nv_p\left( ab \right) + \exrootscountbound
      .
    \end{align*}

    Assume now that the Newton polygon of $f$ consists of two segments of different slopes and that $v\left( k \right) = 0$.
    Then, according to condition \ref{item:k_es} of Corollary \ref{cor:explicit_non_ex_bound}, for each non \ex root $\alpha$ of $f$ whose valuation is equal to minus the slope of the rightmost segment of the Newton polygon of $f$,
    \[
      v_p\left( \Deta \right) 
      \leq
      v_p\left( \alpha f'\left( \alpha \right) \right)
      \leq
      v_p\left( b \right) + \frac{0}{n - k}v_p\left( a \right)
      =
      v_p\left( b \right)
    \]
    and according to condition \ref{item:non_k_es} of Corollary \ref{cor:explicit_non_ex_bound} for each non \ex root $\alpha$ of $f$ whose valuation is equal to minus the slope of the leftmost segment of the Newton polygon of $f$,
    \[
      v_p\left( \Deta \right) 
      \leq
      v_p\left( \alpha f'\left( \alpha \right) \right)
      \leq
      \frac{k v_p\left( b \right) - 0 v_p\left( a \right)}{k - 0}
      =
      v_p\left( b \right)
    \]
    Therefore, according to Corollary \ref{cor:ex_bound},    
    \begin{align*}
      v_p\left( \disc\left( L \right) \right)
      =
      &
      \sum_{\alpha} v_p\left( \Deta \right)
      \leq
      \left( n - \exrootscountbound \right) v_p\left( b \right)
      +
      \exrootscountbound \left( \diffbnd \right)
      \leq
      nv_p\left( ab \right) + \exrootscountbound
      .
    \end{align*}


    Assume finally that the Newton polygon of $f$ consists of two segments of different slopes and that $v\left( n - k \right) = 0$.
    Then, according to condition \ref{item:k_es} of Corollary \ref{cor:explicit_non_ex_bound}, for each non \ex root $\alpha$ of $f$ whose valuation is equal to minus the slope of the rightmost segment of the Newton polygon of $f$,
    \[
      v_p\left( \Deta \right) 
      \leq
      v_p\left( \alpha f'\left( \alpha \right) \right)
      \leq
      \frac{
        n v_p\left( a \right) - kv_p\left( 1 \right)
      }{
        n - k
      }
      =
      \frac{n v_p\left( a \right)}{n - k}
    \]
    and according to condition \ref{item:non_k_es} of Corollary \ref{cor:explicit_non_ex_bound} for each non \ex root $\alpha$ of $f$ whose valuation is equal to minus the slope of the leftmost segment of the Newton polygon of $f$,
    \[
      v_p\left( \Deta \right) 
      \leq
      v_p\left( \alpha f'\left( \alpha \right) \right)
      \leq
      v_p\left( 1 \right) + \frac{n}{k} v_p\left( \frac{b}{a } \right)
      =
      \frac{n}{k} v_p\left( \frac{b}{a } \right)
    \]
    Therefore, according to Corollary \ref{cor:ex_bound},    
    \begin{align*}
      v_p\left( \disc\left( L \right) \right)
      =
      &
      \sum_{\alpha} v_p\left( \Deta \right)
      \leq
      \left( n - \exrootscountbound \right) v_p\left( b \right)
      +
      \exrootscountbound \left( \diffbnd \right)
      \leq
      nv_p\left( ab \right) + \exrootscountbound
      .
    \end{align*}
  \end{proof}
