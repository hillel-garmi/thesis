  \section{Local Analysis}
  Let $\F$ be a non-Archimedean local field with local ring $R$, and residue field $\kappa$. 
  Let $\overline{\F}$ be an algebraic closure of $\F$ and let $\overline{\kappa}$ be the residue field of $\overline{\F}$.
  Let $v$ be an extension to $\overline{\F}$ of the unique valuation of $\F$ for which $v\left( K \right) = \Z$.
  % Fix $n,m,k$ pairwise distinct non-negative integers with $\gcd\left( n,m,k \right) = 1$.
  Let 
  \[
    f = a_n x^n + a_m x^m + a_k x^k \in \F\left[ x, x^{-1} \right]
  \]
  be a trinomial with $a_n, a_m, a_k \neq 0$ and 
  \[
    1 = \gcd\left( n - m, m - k, k - n \right) = \gcd\left( n - m, m - k \right).
  \]
  When use $\left( f \right)$ to denote the principal ideal generated by $f$ in the ring $\kappa\left[ x, x^{-1} \right]$.
  Note that no assumption is made about the order of $n,m,k$ and thus all statements about them are true for all of the permutations of $n,m,k$.
  In light of this, in the sequel we often state or define something for some permutaion of $n,m,k$ and then use it for other permutations as well.
  % Additionally, so we do not have to worry about the (maybe negative) power of $x$ factor of $f$, we suggest to identify $f$ with the ideal it generates in $\F\left[ x, x^{-1} \right]$ and interpret 
  
  Let $\alpha \in \overline{\F}^\times$ be a root of $f$.
  Note that 
  \[
    \alpha f'\left( \alpha \right) 
    = 
    \alpha f'\left( \alpha \right) - nf\left( \alpha \right)
    % n a_n \alpha^n + m a_m \alpha^m + k a_k \alpha^k 
    =
    (m - n) a_m \alpha^m + (k - n) a_k \alpha^k.
  \] 
  Thus, let us denote
  \[
    \h{n} 
    =
    xf - nf 
    = 
    (m - n) a_m x^m + (k - n) a_k x^k
  \]
  and so we can speak of each of its terms separately, let us also denote
  \[
    f_{n,m} = (m - n) a_m x^m.
  \]
  For most cases, we can bound $v\left( f'\left( \alpha \right) \right)$ using the non-Archimedean triangle inequality and the term of least valuation of either $\h{n}$, $\h{m}$ or $\h{k}$.
  We will try to first characterize the cases where this strategy fails.
  
  \subsection{Magnetic roots}
  Note that

  % \[

  % \]
  % the inequality
  \begin{align}
    \label{eq:valuation_inequality_condition}
    & v\left( \alpha f'\left( \alpha \right) \right) 
    % =
    % v\left( f_{n,m}\left( \alpha \right) +  f_{n,k}\left( \alpha \right) \right) 
    > 
    \min\left\{  
      v\left( f_{n,m}\left( \alpha \right) \right), 
      v\left( f_{n,k}\left( \alpha \right)\right)
    \right\}
  \end{align}
  % is equivalent to
  if and only if
  \begin{equation}
    \label{eq:valuation_positivity_condition}
    v\left( 1 + \frac{f_{n,m}\left( \alpha \right)}{f_{n,k}\left( \alpha \right)} \right) > 0,
  \end{equation}
  \directorscut{(indeed, in both cases $v\left( f_{n,k}\left( \alpha \right) \right) = v\left( f_{n,m}\left( \alpha \right) \right)$, and hence the equivalence follows by reducing $v\left( f_{n,k}\left( \alpha \right) \right)$ from both sides of the first inequality),}
  i.e. if and only if 
  \begin{equation}
    \label{eq:principal_unit_condition}
    -\frac{f_{n,m}\left( \alpha \right)}{f_{n,k}\left( \alpha \right)}
    \text{ is a principal unit.}
  \end{equation} 
  Since $\F\left[ x, x^{-1} \right]^\times = \F^\times x^{\Z}$, if condition \ref{eq:principal_unit_condition} holds for $f$ then it also holds for any polynomial in the ideal $\left( f \right)$.
  Thus, we can make the following
  % Note that equivelently, 
  % \[
  %   \left( \frac{f_{m,n}\left( \alpha \right)}{f_{n,k}\left( \alpha \right)} \right)^{-1} =

  \begin{definition}
    If the equivalent conditions \ref{eq:valuation_inequality_condition}, \ref{eq:valuation_positivity_condition} and \ref{eq:principal_unit_condition} hold, we say that $\alpha$ is an \textbf{\es{n}} root of $\left( f \right)$ or simply of $f$.
    % , or just \es{n} when $f$ is clear from context
    We say that $\alpha$ is \textbf{\anex} root of $\left( f \right)$,
    or simply of $f$,
    % or just \anex root when $f$ is clear from context
    if it is an \es{n}, \es{m} and \es{k} root of $f$.
  \end{definition}

  \begin{lemma}
    \label{lem:two_escaping_implies_ex}
    If $\alpha$ is \es{n} and \es{m}, then $\alpha$ is \ex.
  \end{lemma}

    Since $-\frac{
        f_{n,m}\left( \alpha \right)
      }{
        f_{n,k}\left( \alpha \right)
      }$ and $-\frac{
        f_{m,k}\left( \alpha \right)
      }{  
        f_{m,n}\left( \alpha \right)
      }$ are both principal units, it follows that so is
    \begin{align*}
      & 
      \left( -\frac{
        f_{n,m}\left( \alpha \right)
      }{
        f_{n,k}\left( \alpha \right)
      } \right)
      \left( -\frac{
        f_{m,k}\left( \alpha \right)
      }{  
        f_{m,n}\left( \alpha \right)
      } \right) = 
      -\frac{
        f_{m,k}\left( \alpha \right)
      }{
        f_{n,k}\left( \alpha \right)
      }.
    \end{align*}
    \directorscut{

      \begin{align*}
        & 
        \left( -\frac{
          f_{n,m}\left( \alpha \right)
        }{
          f_{n,k}\left( \alpha \right)
        } \right)
        \left( -\frac{
          f_{m,k}\left( \alpha \right)
        }{  
          f_{m,n}\left( \alpha \right)

    \label{cor:scalability}
    Let $\alpha$ be \anex root of $f$ $a \in \F^\times$ and $\delta =\pm 1$.
    Then $a\alpha^\delta$ is \anex root of $f\left( x^\delta / \alpha \right)$.

  \begin{lemma}
    If $\alpha$ is an \es{n} root of $f$, then $v\left( \alpha \right) \in \frac{1}{m - k} \Z$.
  \end{lemma}
  \begin{proof}
    Follows from the fact that 
    \[
      0 = 
      v\left(  
        \frac{
          f_{n,m}\left( \alpha \right)
        }{
          f_{n,k}\left( \alpha \right)
        }
      \right) =
      v\left( m - n \right) + v\left( a_m \right) - v\left( n - k \right) - v\left( a_k \right) + (m - k) v\left( \alpha \right).
    \]
  \end{proof}
  \begin{corollary}
    \label{cor:integral_valuation}
    If $\alpha$ is \anex root of $f$, then $v\left( \alpha \right) \in \Z$.
  \end{corollary}
  \begin{proof}
    By the previous lemma, 
    \[
      v\left( \alpha \right) \in
      \frac{1}{m - n} \Z \cap \frac{1}{n - k} \Z =
      \frac{1}{\gcd\left( m - n, n - k \right)} \Z =
      \Z.
    \]
  \end{proof}
  
  \begin{corollary}
    If $\alpha$ is \anex root of $f$, then there is a $h \in \F^\times x^{\pm 1}$ such that $v\left( h\left( \alpha \right) \right) = 0$ and $f \circ h^{-1}$ is integral over $R$.
  \end{corollary}
  \begin{proof}
    By Corollary \ref{cor:integral_valuation}, $v\left( \alpha \right) \in \Z$, and hence we can choose $a \in K \cap v^{-1}\left( -v\left( \alpha \right) \right)$.
    Then $v\left( a\alpha \right) = 0$.
    Since $f\left( a^{-1}x \right)$ is a trinomial, its Newton polygon has at most two segments, and hence the set of valuations of roots is of cardinality at most $2$.
    Hence, as we have already shown that $0$ is in the set of valuations of roots of $f$, it follows that this set is either non negative or non positive.
    We can set $\delta = 1$ in the first plave and $\delta = -1$ in the second place (and either of them if both cases hold) and verify that the corollary holds for $h = a x^\delta$.
  \end{proof}

  \end{definition}

  \begin{lemma}
    \label{lem:slope_length_valuation}
    Assume that $f$ is \anredex{n} polynomial.
    Then
    \begin{enumerate}
      \item \label{item:smaller_slope_length_valuation} 
      $v\left( n - k \right) = v\left( n - m \right) = 0$.
      \item \label{item:bigger_slope_length_valuation} 
      $v\left( m - k \right) 
        = 
        v\left( a_n \right) - v\left( a_m \right)
        % =
        % v\left( a_n \right) - v\left( a_k \right)
        .$
    \end{enumerate}
  \end{lemma}
  \begin{proof}
    Assume that $\alpha$ is one of the \ex roots of $f$ with valuation $0$.
    The equalities 
    \[
      v\left( f_{k,n}\left( \alpha \right) \right) = 
      v\left( f_{k,m}\left( \alpha \right) \right)
      \quad \text{and} \quad
      v\left( f_{m,n}\left( \alpha \right) \right) = 
      v\left( f_{m,k}\left( \alpha \right) \right)
    \]
    imply respectively that
    \begin{gather}
      v\left( m - k \right) = v\left( n - k \right) + v\left( a_n \right) - v\left( a_m \right) 
      \label{eq:first_relation_between_slope_length_valuations}
      \\
      \notag
      \text{ and } 
      \\
      v\left( m - k \right) = v\left( n - m \right) + v\left( a_n \right) - v\left( a_k \right) 
      \label{eq:second_relation_between_slope_length_valuations}
    \end{gather}
    However, since $v\left( a_m \right) = v\left( a_k \right)$, these two equalities imply together that $v\left( n - k \right) = v\left( n - m \right)$.
    Since $1 = \gcd\left( n - m, n - k \right)$, it follows that $v\left( n - k \right) = v\left( n - m \right) = 0$, proving item \ref{item:smaller_slope_length_valuation}.
    Substituting back in either equation \ref{eq:first_relation_between_slope_length_valuations} or equation \ref{eq:second_relation_between_slope_length_valuations} proves item \ref{item:bigger_slope_length_valuation}.
  \end{proof}
  In light of the previous lemma, we make the following:
  \begin{definition}
    We say that $f$ is an \textbf{\nredir{n}} polynomial if 
    \begin{enumerate}
      \item 
      % \label{item:smaller_slope_length_valuation} 
      $v\left( n - k \right) = v\left( a_m \right) = v\left( n - m \right) = v\left( a_k \right) = 0$.
      \item 
      % \label{item:bigger_slope_length_valuation} 
      $v\left( m - k \right) 
        = 
        v\left( a_n \right)
        .$
    \end{enumerate} 
  \end{definition}

  Assume now that $f$ is an \nredir{n} polynomial.
  Since $1 = \gcd\left( n - m, n - k \right)$, there is a unique $u \in \kappa^\times$ such that
  \begin{equation}
    \label{eq:unique_u}
    u^{n - m} = \frac{m - k}{k - n}\frac{a_m}{a_n} \in \kappa^\times
    \quad \text{and} \quad
    u^{n - k} = \frac{k - m}{m - n}\frac{a_k}{a_n} \in \kappa^\times.
  \end{equation}
  Thus we make the following
  \begin{definition}
    We define $u\left( f \right)$ to be the unique member of $\kappa^\times$ satisfying condition \ref{eq:unique_u}.

  \begin{lemma}
    \label{lem:magnetic_roots_image}
    Assume that $v\left( \alpha \right) = 0$.
    Then $\alpha$ is a \ex root of $f$ if and only if its image in $\overline{\kappa}^\times$ is $u\left( f \right)$.
  \end{lemma}
  \begin{proof}
    By Lemma \ref{lem:two_escaping_implies_ex}, $\alpha$ is \ex, if and only if it is \es{m} and \es{k}, which is equivalent by Lemma \ref{lem:escaping_image_equivalence} and the choice of $u$ to the image of $\alpha$ in $\overline{\kappa}$ being equal to $u$.
  \end{proof}

  \subsubsection{The equal valuation case \editorial{(think of a better name?)}}
  Assume now that $v\left( a_n \right) = v\left( a_m \right) = v\left( a_k \right) = 0$.

  \begin{lemma}
    \label{lem:magnetism_away_from_ab}
    $\alpha$ is \anex root of $f$ if and only if $v\left( f'\left( \alpha \right) \right) > 0$.
  \end{lemma}
  \begin{proof}
    Since the Newton polygon of $f$ has only segments of slope $0$, $v\left( \alpha \right) = 0$.
    Therefore, since $1 = \gcd\left( m - n, k - n \right)$, it follows that 
    \[
      \min\left\{  
        v\left( f_{n,m}\left( \alpha \right) \right),
        v\left( f_{n,k}\left( \alpha \right) \right)
      \right\} = 0.
    \]
  \end{proof}

  \begin{lemma}
    \label{lem:2_magnetic_roots}
    $f$ has at most $2$ magnetic roots.
  \end{lemma}
  \begin{proof}
    Assume that it has \anex root.
    Then by Lemma \ref{lem:slope_length_valuation}, $f$ is an \nredir{n} polynomial and in particular $v\left( m - k \right) = 0$ and $u\left( f \right) \in \kappa^\times$ can be defined.
    Since all of the slopes of the Newton polygon of $f$ are of slope $0$, all of its roots are of valuation $0$.
    Therefore, by Lemma \ref{lem:magnetic_roots_image} all of the magnetic roots of $f$ lie in the fiber of $u\left( f \right)$ and thus it suffices to show that $u\left( f \right)$ is not a triple root of $f$.
    However if it was it would have been a double root of 
    \[
      f_n 
      = 
      xf' - nf 
      =
      x^k \left( a_m x^{m - k} + a_k \right) 
    \] 
    contradicting the reducedness of $\left( a_m x^{m - k} + a_k \right) \leq \kappa\left[ x, x^{-1} \right]$ which follows in turn form the mentioned fact that $v\left( m - k \right) = 0$.
  \end{proof}

  \begin{corollary}
    The inertia group acts on the roots of $f$ either trivially or as a single transposition.
  \end{corollary}
  \begin{proof}
    It suffices to show that $f$ has at most $2$ roots $\alpha$ at which $v\left( f'\left( \alpha \right) \right) > 0$, or equivalently, according to Lemma \ref{lem:magnetism_away_from_ab}, to show that $f$ has at most two \ex roots, which is true by Lemma \ref{lem:2_magnetic_roots}. 
  \end{proof}
\subsubsection{The non equal valuation case \editorial{(think of a better name?)}}
  Assume now that $f$ is an \nredir{n} polynomial with 
  \[
    v\left( m - k \right) = v\left( a_n \right) > 0.
  \]
  It follows in particular that $p := {\rm char}\left( \kappa \right) > 0$ and we assume moreover that $\kappa$ is perfect.
  \begin{lemma}
    \label{lem:counting_ex_roots}
    $f$ has exactly $p^{v_p\left( m - k \right)}$ \ex roots of valuation $0$ (in particular, $f$ is \ex).
  \end{lemma}
  \begin{proof}
    The fact that $u\left( f \right)$ satisfies condition \ref{eq:unique_u} by definition, can be utilized
    \directorscut{
      as follows
      \begin{align*}
        & 
        u\left( f \right)^{-n}f\left( u\left( f \right) \right) 
        = 
        a_m u\left( f \right)^{m-n} + a_k u\left( f \right)^{k-n} 
        =
        a_n \frac{k - n}{m - k} + a_n \frac{m - n}{k - m} 
        =
        -a_n
        =
        0 \in \kappa, 
      \end{align*}
    }
    to show that it is a root of $f$ in $\kappa$.
    Moreover, we can show that the multiplicity of $u\left( f \right)$ as a root of $f$ is exactly $p^{v_p\left( m - k \right)}$ since 
    \[

      = 
      x^k \left( 
        a_m x^{m - k} + a_k 
      \right) 
      =

        a_m^{\left( p^{-v_p\left( m - k \right)} \right)} x^{\frac{m - k}{p^{v_p\left( m - k \right)}}} + a_k^{\left( p^{-v_p\left( m - k \right)} \right)}
      \right)^{p^{v_p\left( m - k \right)}}
      \in \kappa\left[ x \right]
    \]
    and $\left( a_m^{\left( p^{-v_p\left( m - k \right)} \right)} x^{\frac{m - k}{p^{v_p\left( m - k \right)}}} + a_k^{\left( p^{-v_p\left( m - k \right)} \right)} \right) \leq \kappa\left[ x, x^{-1} \right]$ is reduced.

  \end{proof}
  We denote by $\D_\alpha$ the different ideal of $\F\left( \alpha \right)$ and deduce from Lemma \ref{lem:counting_ex_roots} the following
  \begin{corollary}
    If $\alpha$ is a magnetic root of $f$ with valuation $0$, then
    $$v\left( \D_\alpha \right) \leq 1 - p^{-v_p\left( a_n \right)} + v_p\left( a_n \right).$$
  \end{corollary}
  \begin{proof}
    By Lemma \ref{lem:magnetic_roots_image} the \ex roots, 
    the image of $\alpha$ in $\overline{\kappa}$ is $u\left( f \right)$, and so is the image of any other \ex root of $f$ with valuation $0$.
    However, by Lemma \ref{lem:counting_ex_roots}, $f$ has at most $p^{v_p\left( m - k \right)}$ \ex roots of valuation $0$, and in praticular at most $p^{v_p\left( m - k \right)}$ whose image in $\overline{\kappa}$ is $u\left( f \right)$.
    Therefore, the ramification degree of $\alpha$ is at most $p^{v_p\left( m - k \right)}$.
    Denote this ramification degree by $e$.
    Then, $1-e^{-1} + v_p\left( e \right)$ is a well known upper bound on the valuation of $\Deta$, the different of $\Q_p\left( \alpha \right)$ over $\Q_p$ (see for example \cite[Remark 1 to Proposition 13 of Chapter. III \S 7]{SerreLocalFields}).
    However, $1 - e^{-1} + v_p\left( e \right) \leq 1 - p^{-\maxpval} + \maxpval$ and hence the result follows. 
  \end{proof}

  \subsection{The derivative at non \ex roots}
  \begin{lemma}
    \label{lem:n_es_bound}
    If $\alpha$ is a non \es{n} root of $f$ then
    \[
      v\left( \alpha f'\left( \alpha \right) \right) 
      \leq
      \max\left\{ a_m \alpha^m, a_k \alpha^k \right\} 
    \]
  \end{lemma}
  \begin{proof}
    From the definition of \es{n} roots it follows that 
    \[
      v\left( \alpha f'\left( \alpha \right) \right) 
      \leq
      \min\left\{ \left( m - n \right)a_m \alpha^m, \left( k - n \right)a_k \alpha^k \right\}
    \]
    and hence the result follows from the assumption that 
    $
      1 = \gcd\left( m - n, k - n \right)
    $
    and hence either $v\left( m-n \right) = 0$ in which case $v\left( \alpha f'\left( \alpha \right) \right) \leq v\left( a_m \alpha^m \right)$ or $v\left( k - n \right) = 0$ in which case $v\left( \alpha f'\left( \alpha \right) \right) \leq v\left( a_k \alpha^k \right)$.
  \end{proof}
  \begin{corollary}
    If $\alpha$ is a non \ex root of $f$ then
    \[
      v\left( \alpha f'\left( \alpha \right) \right) 
      \leq
      \max\left\{  
        v\left( a_n \alpha^n \right),
        v\left( a_m \alpha^m \right),
        v\left( a_k \alpha^k \right)
      \right\}.
    \]
  \end{corollary}
  \begin{proof}
    By Lemma \ref{lem:two_escaping_implies_ex} we can assume without loss of generality that $\alpha$ is neither an \es{n} root of $f$ nor an \es{m} root of $f$.
    Therefore, it follows from Lemma \ref{lem:n_es_bound} that 
    \begin{align*}
      v\left( \alpha f'\left( \alpha \right) \right)
      \leq
      &
      \min\left\{  
        \max\left\{  
          a_m \alpha^m,
          a_k \alpha^k
        \right\},  
        \max\left\{  
          a_n \alpha^n,
          a_k \alpha^k
        \right\}
      \right\}
      \leq \\
      & \max\left\{  
        \max\left\{  
          a_m \alpha^m,
          a_k \alpha^k
        \right\},  
        \max\left\{  
          a_n \alpha^n,
          a_k \alpha^k
        \right\}
      \right\} \leq \\
      & \max\left\{  
        a_n \alpha^n,
        a_m \alpha^m,
        a_k \alpha^k
      \right\}
    \end{align*} 
  \end{proof}

  Assume now that $f = x^n + ax^k + b$.
