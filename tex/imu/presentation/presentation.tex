    \only<1-4>{
        \begin{block}{}
            \[
                \Gal\left( f \right) :=
                \Gal\left( {\rm Spl}\left( f \right) \right) \hookrightarrow
                {\rm Sym}\left( \left\{ \text{Roots of } f \right\} \right) \simeq
                S_n
            \]
        \end{block}
    }
    \only<2-6>{
        \begin{alertblock}{\only<2-5>{Hilbert's Irreducibility Theorem (1892)}}
            For "almost" any polynomial $f \in \Q[x]$ of degree $n$, $\Gal(f) = S_n$.
        \end{alertblock}
    }
    \only<3-8>{
        \begin{alertblock}{\only<3-6>{Schur (1931)}}
            \only<3-7>{The Galois group of} 
            \only<3-7>{\[
                f =\sum_{k=0}^n \binom{n}{k} \frac{x^k}{k!}
            \]}
            is $S_n$ for all $n \geq 1$.
        \end{alertblock}
    }
    \only<4-9>{
        \begin{alertblock}<4->{Nart \& Vila (1979) }
            The Galois group of $x^n - x- 1$ is $S_n$ for all $n \geq 2$.
        \end{alertblock}
    }
    % \begin{alertblock}<4->{Bary-Soroker \& Kozma (2020) }
    %     If we sample the coefficients of a monic polynomial of degree $n$ uniformly and independently from $\left\{ 1,\dots,210 \right\}$ then $\PP\left( \Gal\left( f \right) \in \left\{ A_n, S_n \right\} \right) \underset{n \to \infty}{\longrightarrow} 1$.
    % \end{alertblock}
    \only<5-9>{
        \begin{block}{}
            \only<5-9>{\[
                f = x^n - 2x + 1
            \]}
            \only<6-9>{
                \[
                    f = \left( x - 1 \right)g
                \]
            }
            \only<7-9>{
                \[
                    g = x^{n-1} + x^{n-2} + \dots + x - 1
                \]
            }
        \end{block}
    }
    \only<8-12>{
        \begin{alertblock}{\only<4-10>{Miles (1960) \& Wolfram (1998)}}
            $g$ is irreducible over $\Q$ for all $n \geq 2$.
        \end{alertblock}
    }
    \only<9->{
        \begin{alertblock}{Martin (2004)}
            The Galois group of $g$ is $S_{n-1}$ for any odd $n$.
        \end{alertblock}
    }
    \only<10->{
        \begin{alertblock}{G. (2026)}
            Let $a,b \in \Z_{\neq 0}$.
            If  
            \[
                \left( a,b \right) \in \left\{
                    \begin{matrix}
                        \left( \pm 1, \pm 1\right),
                        \left( \pm 1, \pm 2\right), 
                        \left( \pm 2, \pm 1\right), 
                        \left( \pm 2, \pm 2\right), \\ 
                        \left( \pm 1, \pm 3 \right),
                        \left( \pm 3, \pm 1 \right),
                        \left( \pm 3, - 3 \right)
                    \end{matrix}
                \right\}
            \]
            then for any $n > 3$, the Galois group of $f = x^n + ax + b$ is the largest that its factorization type allows.
        \end{alertblock}
    }

    \begin{alertblock}<2->{Conjecture / Pre-Theorem}
        For any integers $0 < k < n$, such that ${\rm gcd}\left( n,k \right) = 1$, and $n > 7$, let 
        \[
            f\left( x \right) = x^n \pm 2x^k \pm 1
        \] 
        Then one of two cases occures:
        \begin{itemize}
            \item $f\left( 1 \right)f\left( -1 \right) = 0$ and the Galois group of $f$ is $S_{n-1}$.
            \item $f\left( 1 \right)f\left( -1 \right) \neq 0$ and the Galois group of $f$ is $S_n$.
        \end{itemize}
    \end{alertblock}
